@app.route("/sell", methods=["GET", "POST"]) @login_required def sell(): """Sell shares of stock""" # User reached route via POST (as by submitting a form via POST) if request.method == "POST": symbol = request.form.get("symbol") shares = request.form.get("shares") if not symbol: return apology("must provide symbol", 400) try: shares = int(shares) if shares <= 0: return apology("shares must be a positive integer", 400) except ValueError: return apology("shares must be a positive integer", 400) # Get user's shares for the selected symbol user_shares = db.execute( "SELECT SUM(shares) as total_shares FROM transactions WHERE user_id = ? AND symbol = ? GROUP BY symbol", session["user_id"], symbol.upper() ) if len(user_shares) != 1 or user_shares[0]["total_shares"] < shares: return apology("you do not own that many shares", 400) # Get current stock price stock = lookup(symbol.upper()) if not stock: return apology("invalid symbol", 400) price = stock["price"] total_sale = shares * price # Get user's current cash user_cash = db.execute("SELECT cash FROM users WHERE id = ?", session["user_id"]) current_cash = user_cash[0]["cash"] # Update user's cash db.execute( "UPDATE users SET cash = ? WHERE id = ?", current_cash + total_sale, session["user_id"] ) # Record the transaction (negative shares for selling) db.execute( "INSERT INTO transactions (user_id, symbol, shares, price) VALUES (?, ?, ?, ?)", session["user_id"], symbol.upper(), -shares, price ) return redirect("/") # User reached route via GET (as by clicking a link or via redirect) else: # Get symbols owned by user for select menu stocks = db.execute( "SELECT symbol FROM transactions WHERE user_id = ? GROUP BY symbol HAVING SUM(shares) > 0", session["user_id"] ) return render_template("sell.html", stocks=stocks)
